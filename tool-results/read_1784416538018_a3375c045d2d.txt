    47→  const [saving, setSaving] = React.useState<Record<string, boolean>>({});
    48→
    49→  const reload = React.useCallback(async () => {
    50→    setLoading(true);
    51→    try {
    52→      const [s, g] = await Promise.all([
    53→        adminFetch<SettingsResponse>("/api/admin/settings"),
    54→        adminFetch<GalleryResponse>("/api/admin/gallery"),
    55→      ]);
    56→      setBusiness(s.business);
    57→      setSettings(s.settings);
    58→      setGallery(g.items ?? []);

    60→      toast.error(lang === "en" ? "Failed to load editor data" : "Error al cargar el editor");

    62→    } finally {
    63→      setLoading(false);
    64→    }
    65→  }, [lang]);
    66→

    74→    } catch (e) {

   377→function GalleryManager({
   378→  items, lang, onUpdated, onReload,
   379→}: {
   380→  items: GalleryItem[];
   381→  lang: "es" | "en";
   382→  onUpdated: (next: GalleryItem[]) => void;
   383→  onReload: () => void;
   384→}) {
   385→  const [tab, setTab] = React.useState<GalleryTab>("photos");
   386→  const [editing, setEditing] = React.useState<GalleryItem | null>(null);
   387→  const [creating, setCreating] = React.useState<GalleryTab | null>(null);
   388→
   389→  // existing before/after pairs (for the pair selector)
   390→  const pairs = React.useMemo(() => {
   391→    const map = new Map<string, { pairId: string; before?: GalleryItem; after?: GalleryItem }>();
   392→    for (const it of items) {
   393→      if (!it.isBeforeAfter || !it.pairId) continue;
   394→      const p = map.get(it.pairId) ?? { pairId: it.pairId };
   395→      if (it.pairRole === "before") p.before = it;
   396→      if (it.pairRole === "after") p.after = it;
   397→      map.set(it.pairId, p);
   398→    }
   399→    return Array.from(map.values());
   400→  }, [items]);
   401→
   402→  // items in current tab (sorted)
   403→  const tabItems = React.useMemo(() => {
   404→    let list: GalleryItem[] = items;
   405→    if (tab === "photos") list = items.filter((it) => !it.isBeforeAfter && it.mediaType === "photo");
   406→    if (tab === "before-after") list = items.filter((it) => it.isBeforeAfter);
   407→    if (tab === "videos") list = items.filter((it) => !it.isBeforeAfter && it.mediaType === "video");
   408→    return [...list].sort((a, b) => a.sortOrder - b.sortOrder);
   409→  }, [items, tab]);
   410→
   411→  // CRUD handlers
   412→  const toggleActive = async (item: GalleryItem, active: boolean) => {
   413→    onUpdated(items.map((it) => (it.id === item.id ? { ...it, active } : it)));
   414→    try {
   415→      await adminFetch(`/api/admin/gallery/${item.id}`, { method: "PUT", body: JSON.stringify({ active }) });
   416→      toast.success(active ? (lang === "en" ? "Visible" : "Visible") : (lang === "en" ? "Hidden" : "Oculto"));
   417→    } catch {
   418→      toast.error(lang === "en" ? "Update failed" : "Error al actualizar");
   419→      onReload();
   420→    }
   421→  };
   422→
   423→  const handleDelete = async (item: GalleryItem) => {
   424→    if (!window.confirm(lang === "en" ? "Delete this gallery item?" : "¿Eliminar este item de galería?")) return;
   425→    onUpdated(items.filter((it) => it.id !== item.id));
   426→    try {
   427→      await adminFetch(`/api/admin/gallery/${item.id}`, { method: "DELETE" });
   428→      toast.success(lang === "en" ? "Item deleted" : "Item eliminado");
   429→    } catch {
   430→      toast.error(lang === "en" ? "Delete failed" : "Error al eliminar");
   431→      onReload();
   432→    }
   433→  };
   434→
   435→  const move = async (item: GalleryItem, dir: -1 | 1) => {
   436→    const list = tabItems;
   437→    const idx = list.findIndex((it) => it.id === item.id);
   438→    const newIdx = idx + dir;
   439→    if (idx === -1 || newIdx < 0 || newIdx >= list.length) return;
   440→    const reordered = [...list];
   441→    const tmp = reordered[idx];
   442→    reordered[idx] = reordered[newIdx];
   443→    reordered[newIdx] = tmp;
   444→    const renumbered = reordered.map((it, i) => ({ ...it, sortOrder: i }));
   445→    // optimistic: patch sortOrder for those items in the parent `items` array
   446→    const map = new Map(renumbered.map((it) => [it.id, it.sortOrder]));
   447→    onUpdated(items.map((it) => (map.has(it.id) ? { ...it, sortOrder: map.get(it.id)! } : it)));
   448→    try {
   449→      await Promise.all(
   450→        renumbered.map((it) =>
   451→          adminFetch(`/api/admin/gallery/${it.id}`, { method: "PUT", body: JSON.stringify({ sortOrder: it.sortOrder }) }),
   452→        ),
   453→      );
   454→      toast.success(lang === "en" ? "Order saved" : "Orden guardado");
   455→    } catch {
   456→      toast.error(lang === "en" ? "Save failed" : "Error al guardar");
   457→      onReload();
   458→    }
   459→  };
   460→
   461→  return (
   462→    <div className="flex flex-col gap-4">
   463→      <Tabs value={tab} onValueChange={(v) => setTab(v as GalleryTab)}>
   464→        <TabsList>
   465→          <TabsTrigger value="photos"><ImageIcon className="size-3.5" />{lang === "en" ? "Photos" : "Fotos generales"}</TabsTrigger>
   466→          <TabsTrigger value="before-after">{lang === "en" ? "Before/After" : "Antes y Después"}</TabsTrigger>
   467→          <TabsTrigger value="videos"><Film className="size-3.5" />{lang === "en" ? "Videos" : "Videos"}</TabsTrigger>
   468→        </TabsList>
   469→
   470→        {(["photos", "before-after", "videos"] as GalleryTab[]).map((tb) => {
   471→          const list = tb === "photos"
   472→            ? items.filter((it) => !it.isBeforeAfter && it.mediaType === "photo")
   473→            : tb === "before-after"
   474→              ? items.filter((it) => it.isBeforeAfter)
   475→              : items.filter((it) => !it.isBeforeAfter && it.mediaType === "video");
   476→          const sorted = [...list].sort((a, b) => a.sortOrder - b.sortOrder);
   477→          return (
   478→            <TabsContent key={tb} value={tb} className="mt-3">
   479→              <div className="flex justify-end mb-2">
   480→                <Button size="sm" onClick={() => setCreating(tb)}>
   481→                  <Plus className="size-4" />
   482→                  {lang === "en" ? "Add item" : "Añadir item"}
   483→                </Button>
   484→              </div>
   485→              {sorted.length === 0 ? (
   486→                <EmptyState title={lang === "en" ? "No items yet" : "Sin items"} hint={lang === "en" ? "Click \"Add item\" to upload one." : "Haz clic en \"Añadir item\" para subir uno."} />
   487→              ) : (
   488→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3 max-h-[60vh] overflow-y-auto scroll-thin pr-1">
   489→                  {sorted.map((it, i) => (
   490→                    <GalleryCard
   491→                      key={it.id}
   492→                      item={it}
   493→                      lang={lang}
   494→                      index={i}
   495→                      total={sorted.length}
   496→                      onEdit={() => setEditing(it)}
   497→                      onDelete={() => void handleDelete(it)}
   498→                      onToggle={(v) => void toggleActive(it, v)}
   499→                      onMove={(dir) => void move(it, dir)}
   500→                    />
   501→                  ))}
   502→                </div>
   503→              )}
   504→            </TabsContent>
   505→          );
   506→        })}
   507→      </Tabs>
   508→
   509→      {/* Edit dialog */}
   510→      <GalleryItemDialog
   511→        open={!!editing}
   512→        item={editing}
   513→        mode={editing ? modeOfItem(editing) : "photos"}
   514→        pairs={pairs}
   515→        lang={lang}
   516→        onOpenChange={(o) => { if (!o) setEditing(null); }}
   517→        onSaved={() => { setEditing(null); onReload(); }}
   518→      />
   519→
   520→      {/* Create dialog */}
   521→      <GalleryItemDialog
   522→        open={!!creating}
   523→        item={null}
   524→        mode={creating ?? "photos"}
   525→        pairs={pairs}
   526→        lang={lang}
   527→        onOpenChange={(o) => { if (!o) setCreating(null); }}
   528→        onSaved={() => { setCreating(null); onReload(); }}
   529→      />
   530→    </div>
   531→  );
   532→}

   534→function modeOfItem(item: GalleryItem): GalleryTab {
   535→  if (item.isBeforeAfter) return "before-after";
   536→  if (item.mediaType === "video") return "videos";
   537→  return "photos";

   540→function GalleryCard({
   541→  item, lang, index, total, onEdit, onDelete, onToggle, onMove,

   544→  lang: "es" | "en";
   545→  index: number;
   546→  total: number;

   548→  onDelete: () => void;

   553→  const isBA = item.isBeforeAfter;
   554→  const badgeLabel = isBA
   555→    ? (item.pairRole === "before" ? (lang === "en" ? "Before" : "Antes") : (lang === "en" ? "After" : "Después"))

   560→      <div className="relative aspect-video bg-muted">

   563→            src={item.mediaUrl}
   564→            poster={item.posterUrl ?? undefined}

   566→            muted
   567→            playsInline
   568→            preload="metadata"

   570→        ) : (
   571→          <img src={item.mediaUrl} alt={item.captionEs ?? ""} className={`size-full object-cover ${item.active ? "" : "opacity-40 grayscale"}`} />

   574→          <Badge variant="secondary" className="text-[9px]">{badgeLabel}</Badge>

   576→        </div>
   577→        <div className="absolute top-1.5 right-1.5 flex gap-0.5">
   578→          <button
   579→            onClick={() => onMove(-1)}
   580→            disabled={index === 0}
   581→            className="size-6 rounded bg-black/50 text-white flex items-center justify-center hover:bg-black/70 disabled:opacity-30"

   584→            <ArrowUp className="size-3" />
   585→          </button>
   586→          <button
   587→            onClick={() => onMove(1)}
   588→            disabled={index === total - 1}
   589→            className="size-6 rounded bg-black/50 text-white flex items-center justify-center hover:bg-black/70 disabled:opacity-30"
   590→            aria-label={lang === "en" ? "Move down" : "Bajar"}
   591→          >
   592→            <ArrowDown className="size-3" />

   594→          <button
   595→            onClick={onEdit}
   596→            className="size-6 rounded bg-black/50 text-white flex items-center justify-center hover:bg-black/70"
   597→            aria-label={lang === "en" ? "Edit" : "Editar"}
   598→          >
   599→            <Pencil className="size-3" />
   600→          </button>
   601→          <button
   602→            onClick={onDelete}
   603→            className="size-6 rounded bg-black/50 text-rose-300 flex items-center justify-center hover:bg-rose-900/70"
   604→            aria-label={lang === "en" ? "Delete" : "Eliminar"}
   605→          >
   606→            <Trash2 className="size-3" />
   607→          </button>
   608→        </div>
   609→      </div>
   610→      <div className="p-2 flex items-center justify-between gap-2">
   611→        <p className="text-xs text-muted-foreground truncate flex-1">
   612→          {lang === "en" ? (item.captionEn ?? item.captionEs ?? "—") : (item.captionEs ?? item.captionEn ?? "—")}
   613→        </p>
   614→        <Switch checked={item.active} onCheckedChange={onToggle} aria-label="active" />
   615→      </div>
   616→    </div>
   617→  );

   620→function GalleryItemDialog({
   621→  open, item, mode, pairs, lang, onOpenChange, onSaved,
   622→}: {
   623→  open: boolean;
   624→  item: GalleryItem | null;
   625→  mode: GalleryTab;
   626→  pairs: { pairId: string; before?: GalleryItem; after?: GalleryItem }[];
   627→  lang: "es" | "en";
   628→  onOpenChange: (open: boolean) => void;
   629→  onSaved: () => void;
   630→}) {
   631→  const isVideo = mode === "videos" || (item?.mediaType === "video") === true;
   632→  const isBA = mode === "before-after" || (item?.isBeforeAfter ?? false);
   633→
   634→  const [mediaUrl, setMediaUrl] = React.useState<string | null>(null);
   635→  const [posterUrl, setPosterUrl] = React.useState<string | null>(null);
   636→  const [captionEs, setCaptionEs] = React.useState("");
   637→  const [captionEn, setCaptionEn] = React.useState("");
   638→  const [pairMode, setPairMode] = React.useState<"new" | "existing">("new");
   639→  const [pairId, setPairId] = React.useState<string>("");
   640→  const [pairRole, setPairRole] = React.useState<"before" | "after">("before");
   641→  const [active, setActive] = React.useState(true);
   642→  const [saving, setSaving] = React.useState(false);
   643→
   644→  React.useEffect(() => {
   645→    if (!open) return;
   646→    setMediaUrl(item?.mediaUrl ?? null);
   647→    setPosterUrl(item?.posterUrl ?? null);
   648→    setCaptionEs(item?.captionEs ?? "");
   649→    setCaptionEn(item?.captionEn ?? "");
   650→    if (item?.pairId) {
   651→      setPairMode("existing");
   652→      setPairId(item.pairId);
   653→      setPairRole((item.pairRole === "after" ? "after" : "before"));
   654→    } else {
   655→      setPairMode("new");
   656→      setPairId("");
   657→      setPairRole("before");
   658→    }
   659→    setActive(item?.active ?? true);
   660→  }, [open, item]);
   661→
   662→  const onSave = async () => {
   663→    if (!mediaUrl) {
   664→      toast.error(lang === "en" ? "Media is required" : "El medio es obligatorio");
   665→      return;
   666→    }
   667→    if (isBA && pairMode === "existing" && !pairId) {
   668→      toast.error(lang === "en" ? "Pick a pair" : "Elige un par");
   669→      return;
   670→    }
   671→    setSaving(true);
   672→    try {
   673→      const finalPairId = isBA
   674→        ? (pairMode === "new" ? `pair-${Date.now()}-${Math.random().toString(36).slice(2, 6)}` : pairId)
   675→        : null;
   676→      const body = {
   677→        mediaUrl,
   678→        mediaType: (isVideo ? "video" : "photo") as "video" | "photo",
   679→        source: "upload",
   680→        isBeforeAfter: isBA,
   681→        pairId: finalPairId,
   682→        pairRole: isBA ? pairRole : null,
   683→        captionEs: captionEs.trim() || null,
   684→        captionEn: captionEn.trim() || null,
   685→        posterUrl: isVideo ? (posterUrl ?? null) : null,
   686→        sortOrder: item?.sortOrder ?? 0,
   687→        active,
   688→      };
   689→      if (item) {
   690→        await adminFetch(`/api/admin/gallery/${item.id}`, { method: "PUT", body: JSON.stringify(body) });
   691→        toast.success(lang === "en" ? "Item updated" : "Item actualizado");
   692→      } else {
   693→        await adminFetch("/api/admin/gallery", { method: "POST", body: JSON.stringify(body) });
   694→        toast.success(lang === "en" ? "Item created" : "Item creado");
   695→      }
   696→      onSaved();
   697→    } catch (e) {
   698→      toast.error(lang === "en" ? "Save failed" : "Error al guardar");
   699→      void e;
   700→    } finally {
   701→      setSaving(false);
   702→    }
   703→  };
   704→
   705→  return (
   706→    <Dialog open={open} onOpenChange={onOpenChange}>
   707→      <DialogContent className="sm:max-w-xl max-h-[90vh] overflow-y-auto scroll-thin">
   708→        <DialogHeader>
   709→          <DialogTitle>
   710→            {item
   711→              ? (lang === "en" ? "Edit gallery item" : "Editar item de galería")
   712→              : (lang === "en" ? "Add gallery item" : "Añadir item de galería")}
   713→          </DialogTitle>
   714→          <DialogDescription>
   715→            {isBA
   716→              ? (lang === "en" ? "Before/After pair photo." : "Foto de par Antes/Después.")
   717→              : isVideo
   718→                ? (lang === "en" ? "Video reel." : "Video reel.")
   719→                : (lang === "en" ? "General gallery photo." : "Foto general de galería.")}
   720→          </DialogDescription>
   721→        </DialogHeader>
   722→
   723→        <div className="flex flex-col gap-4">
   724→          <MediaDropzone
   725→            label={isVideo ? (lang === "en" ? "Video file" : "Archivo de video") : (lang === "en" ? "Photo" : "Foto")}
   726→            value={mediaUrl}
   727→            mediaType={isVideo ? "video" : "image"}
   728→            onChange={(url) => setMediaUrl(url)}
   729→          />
   730→
   731→          {isVideo ? (
   732→            <MediaDropzone
   733→              label={lang === "en" ? "Poster image (optional)" : "Póster (opcional)"}
   734→              value={posterUrl}
   735→              mediaType="image"
   736→              onChange={(url) => setPosterUrl(url)}
   737→              compact
   738→            />
   739→          ) : null}
   740→
   741→          {isBA ? (
   742→            <div className="flex flex-col gap-3 rounded-md border border-border p-3 bg-muted/30">
   743→              <Label className="text-xs font-semibold">{lang === "en" ? "Pair assignment" : "Asignación de par"}</Label>
   744→              <RadioGroup
   745→                value={pairMode}
   746→                onValueChange={(v) => setPairMode(v as "new" | "existing")}
   747→                className="grid grid-cols-2 gap-2"
   748→              >
   749→                <label className="flex items-center gap-2 text-sm cursor-pointer">
   750→                  <RadioGroupItem value="new" />
   751→                  {lang === "en" ? "New pair" : "Par nuevo"}
   752→                </label>
   753→                <label className="flex items-center gap-2 text-sm cursor-pointer">
   754→                  <RadioGroupItem value="existing" disabled={pairs.length === 0} />
   755→                  {lang === "en" ? "Existing pair" : "Par existente"}
   756→                </label>
   757→              </RadioGroup>
   758→              {pairMode === "existing" ? (
   759→                <div className="flex flex-col gap-1.5">
   760→                  <Label className="text-xs">{lang === "en" ? "Select pair" : "Selecciona par"}</Label>
   761→                  <select
   762→                    value={pairId}
   763→                    onChange={(e) => setPairId(e.target.value)}
   764→                    className="w-full rounded-md border border-input bg-background px-2.5 py-1.5 text-sm focus:outline-none focus:ring-1 focus:ring-ring"
   765→                  >
   766→                    <option value="">{lang === "en" ? "— Pick —" : "— Elegir —"}</option>
   767→                    {pairs.map((p) => (
   768→                      <option key={p.pairId} value={p.pairId}>
   769→                        {p.pairId.slice(0, 12)}… ({p.before ? "✓" : "—"}/{p.after ? "✓" : "—"})
   770→                      </option>
   771→                    ))}
   772→                  </select>
   773→                </div>
   774→              ) : null}
   775→              <div className="flex flex-col gap-1.5">
   776→                <Label className="text-xs">{lang === "en" ? "This photo is the…" : "Esta foto es el…"}</Label>
   777→                <RadioGroup
   778→                  value={pairRole}
   779→                  onValueChange={(v) => setPairRole(v as "before" | "after")}
   780→                  className="grid grid-cols-2 gap-2"
   781→                >
   782→                  <label className="flex items-center gap-2 text-sm cursor-pointer">
   783→                    <RadioGroupItem value="before" />
   784→                    {lang === "en" ? "Before" : "Antes"}
   785→                  </label>
   786→                  <label className="flex items-center gap-2 text-sm cursor-pointer">
   787→                    <RadioGroupItem value="after" />
   788→                    {lang === "en" ? "After" : "Después"}
   789→                  </label>
   790→                </RadioGroup>
   791→              </div>
   792→            </div>
   793→          ) : null}
   794→
   795→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   796→            <div className="flex flex-col gap-1.5">
   797→              <Label>{lang === "en" ? "Caption (ES)" : "Descripción (ES)"}</Label>
   798→              <Input value={captionEs} onChange={(e) => setCaptionEs(e.target.value)} />
   799→            </div>
   800→            <div className="flex flex-col gap-1.5">
   801→              <Label>Caption (EN)</Label>
   802→              <Input value={captionEn} onChange={(e) => setCaptionEn(e.target.value)} />
   803→            </div>
   804→          </div>
   805→
   806→          <div className="flex items-center gap-2">
   807→            <Switch id="gi-active" checked={active} onCheckedChange={setActive} />
   808→            <Label htmlFor="gi-active">{lang === "en" ? "Visible on site" : "Visible en el sitio"}</Label>
   809→          </div>
   810→        </div>
   811→
   812→        <DialogFooter>
   813→          <DialogClose asChild>
   814→            <Button variant="outline" disabled={saving}>{lang === "en" ? "Cancel" : "Cancelar"}</Button>
   815→          </DialogClose>
   816→          <Button onClick={() => void onSave()} disabled={saving}>
   817→            {saving ? <Spinner className="mr-2" /> : null}
   818→            {item ? (lang === "en" ? "Save" : "Guardar") : (lang === "en" ? "Create" : "Crear")}
   819→          </Button>
   820→        </DialogFooter>
   821→      </DialogContent>
   822→    </Dialog>
   823→  );
   824→}

   826→function HoursEditor({

   854→          );
   855→        })}
   856→      </div>
